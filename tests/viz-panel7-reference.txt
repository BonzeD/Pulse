// Pulse Viz 2 — расширяемый движок визуализаций.
// Каждый стиль регистрируется через Viz.register({ id, name, icon, cover, draw(g) }).
// g — кадр: размеры, спектр, бас, бит, ТЕМП (pos/phase/bpm) и помощники для цвета.
// Всё движение привязано к темпу трека: pos — номер доли (дробный), phase — фаза внутри доли.
// makeViz(opts) — отдельный экземпляр движка (предпросмотр в настройках, музыкальные обои).
// opts.ext — функция, возвращающая стили основного экземпляра (чтобы плагины работали и в превью).
const makeViz = (opts = {}) => {
  const REG = []; let onChange = null; const EXT = opts.ext || null;
  let cv, ctx, W = 0, H = 0, pal = [[255, 45, 149], [139, 61, 255], [0, 229, 255]];
  const NB = 64, bands = new Float32Array(NB), peaks = new Float32Array(NB);
  const state = {}; let lastNow = performance.now(), synth = 0, SENS = 1, SR = 48000, level = 0;
  // 1.9.5: синхронизация эффектов с песней — огибающая удара, энергия куплета/припева, «дроп»
  const SY = { prevKick: 0, env: 0, flash: 0, sx: 0, sy: 0, slow: 0.15, fast: 0, drop: 0, lastHit: 0 };
  let LW = 1, GLOW = 1, DENS = 1, HUE = 0, REACT = 0.6, BASSK = 1, COVER = null;
  const TAU = Math.PI * 2;
  let ro = null;
  // 2.0.6: свой WebGL-холст для 3D-стилей; кадр с него копируется в основной холст (копия GPU → GPU)
  let G = null, gcv = null, gpuOff = false;
  function gpuFor() {
    if (gpuOff || opts.gpu === false || typeof PulseGPU === 'undefined') return null;
    if (!G) { gcv = document.createElement('canvas'); G = PulseGPU.create(gcv); if (!G) { gpuOff = true; return null; } }
    return G.ok() ? G : null;
  }
  function init(c) { if (ro) try { ro.disconnect(); } catch {} cv = c; ctx = c.getContext('2d', { alpha: true, desynchronized: true }); resize(); ro = new ResizeObserver(resize); ro.observe(c.parentElement); }
  function destroy() { if (ro) try { ro.disconnect(); } catch {} ro = null; cv = null; ctx = null; W = 0; }
  let blankCv = false; // 2.0.11: холст уже чистый — не стираем повторно
  let lowQ = false, autoLow = false, cost = 0, goodSince = 0, flips = 0;
  function resize() { if (!cv) return; const d = lowQ ? 1 : Math.min(opts.maxDpr || 1.5, devicePixelRatio || 1), r = cv.getBoundingClientRect(); W = r.width; H = r.height; cv.width = Math.max(1, W * d); cv.height = Math.max(1, H * d); ctx.setTransform(d, 0, 0, d, 0, 0); }
  const rgba = (p, a) => `rgba(${p[0] | 0},${p[1] | 0},${p[2] | 0},${Math.max(0, Math.min(1, a))})`;
  function col(f) { f = ((((f + HUE) % 1) + 1) % 1) * pal.length; const i = Math.floor(f), t = f - i, p = pal[i % pal.length], q = pal[(i + 1) % pal.length]; return [p[0] + (q[0] - p[0]) * t, p[1] + (q[1] - p[1]) * t, p[2] + (q[2] - p[2]) * t]; }
  const mix = (f, a = 1) => rgba(col(f), a);
  const ease = (p) => 1 - Math.pow(1 - p, 3);
  const avg = (a, b) => { let s = 0; for (let i = a; i < b; i++) s += bands[i]; return s / (b - a); };
  const bass = () => avg(0, 5), mid = () => avg(8, 30), high = () => avg(34, 60);
  const band = (f) => bands[Math.max(0, Math.min(NB - 1, Math.floor(f * (NB - 1))))];
  function rr(x, y, w, h, r) { r = Math.max(0, Math.min(r, h / 2, w / 2)); ctx.beginPath(); ctx.moveTo(x + r, y); ctx.arcTo(x + w, y, x + w, y + h, r); ctx.arcTo(x + w, y + h, x, y + h, r); ctx.arcTo(x, y + h, x, y, r); ctx.arcTo(x, y, x + w, y, r); ctx.closePath(); }
  // гладкая замкнутая кривая через точки (Catmull-Rom → Безье)
  function smoothClosed(pts) {
    const n = pts.length; if (n < 3) return; ctx.beginPath(); ctx.moveTo(pts[0][0], pts[0][1]);
    for (let i = 0; i < n; i++) {
      const p0 = pts[(i - 1 + n) % n], p1 = pts[i], p2 = pts[(i + 1) % n], p3 = pts[(i + 2) % n];
      ctx.bezierCurveTo(p1[0] + (p2[0] - p0[0]) / 6, p1[1] + (p2[1] - p0[1]) / 6, p2[0] - (p3[0] - p1[0]) / 6, p2[1] - (p3[1] - p1[1]) / 6, p2[0], p2[1]);
    }
    ctx.closePath();
  }
  function glow(c, b) { if (lowQ || GLOW <= 0) return; ctx.shadowColor = c; ctx.shadowBlur = b * GLOW; }
  const noGlow = () => { ctx.shadowBlur = 0; };

  function fillBands(freq, playing, g, dt) {
    const fk = Math.max(0.25, Math.min(4, (dt || 1 / 60) * 60));
    if (freq) {
      const N = freq.length, hz = SR / 2 / N;
      for (let i = 0; i < NB; i++) {
        const f0 = 32 * Math.pow(14000 / 32, i / NB), f1 = 32 * Math.pow(14000 / 32, (i + 1) / NB);
        const a = Math.max(1, Math.floor(f0 / hz)), b = Math.max(a + 1, Math.ceil(f1 / hz));
        let m = 0; for (let k = a; k < b && k < N; k++) m = Math.max(m, freq[k]);
        const v = Math.min(1.15, Math.pow(m / 255, 1.6) * (0.75 + i / 110) * SENS * (i < 8 ? BASSK : 1));
        // атака/спад зависят от «отзывчивости»: чем выше, тем резче полосы попадают в удары
        // 1.9.6: коэффициенты пересчитываются по времени кадра — на 144 Гц полосы больше не «дрожат», на 30 FPS не запаздывают
        const kk = v > bands[i] ? 0.55 + REACT * 0.45 : 0.1 + REACT * 0.3;
        bands[i] += (v - bands[i]) * (kk >= 1 ? 1 : 1 - Math.pow(1 - kk, fk));
      }
    } else {
      // нет доступа к звуку (встроенный YouTube): «синтетический» спектр, но строго в темп
      synth += dt * (g.bpm || 100) / 60;
      for (let i = 0; i < NB; i++) {
        const base = (1 - i / 75) * (0.16 + 0.1 * Math.sin(synth * 1.7 + i * 0.35) + 0.06 * Math.sin(synth * 0.6 - i * 0.12)) + g.beat * 0.55 * Math.pow(1 - i / NB, 2);
        const v = playing ? Math.max(0, base * SENS) : 0.02; bands[i] += (v - bands[i]) * (1 - Math.pow(0.7, fk));
      }
    }
    let s = 0; for (let i = 0; i < NB; i++) { peaks[i] = Math.max(bands[i], peaks[i] - 0.6 * dt); s += bands[i]; }
    level += (s / NB - level) * (1 - Math.pow(0.8, fk));
  }

  function draw(o) {
    if (!ctx) return; if (!W) resize(); if (!W) return;
    const now = performance.now(), dt = Math.min(0.05, (now - lastNow) / 1000); lastNow = now;
    SENS = o.sens || 1; if (o.sr) SR = o.sr;
    const Q = o.quality || 'auto', want = Q === 'eco' ? true : Q === 'high' ? false : autoLow; if (want !== lowQ) { lowQ = want; resize(); }
    const T = o.tempo || {}; const bpm = T.bpm || 100, pos = T.pos != null ? T.pos : (now / 1000) * bpm / 60, phase = ((pos % 1) + 1) % 1;
    const pulse = T.conf ? Math.exp(-phase * 6) * T.conf : 0;
    const beat = o.playing ? Math.max(o.beat || 0, pulse * 0.85) : (o.beat || 0) * 0.5;
    const g0 = { bpm, beat };
    const FX = o.fx || {};
    LW = (FX.thick ?? 100) / 100; GLOW = (FX.glow ?? 100) / 100; DENS = (FX.density ?? 100) / 100; BASSK = (FX.bass ?? 100) / 100; COVER = o.cover || null;
    HUE = (HUE + dt * ((FX.hue || 0) / 100) * 0.06 * (bpm / 120)) % 1;
    fillBands(o.freq, o.playing, g0, dt);
    // удар: только по фронту настоящей атаки (а не постоянная «пульсация»), спад зависит от темпа
    const kick = o.playing ? (o.beat || 0) : 0, spb = 60 / Math.max(60, bpm);
    if (kick > 0.42 && kick > SY.prevKick + 0.18 && now - SY.lastHit > spb * 400) {
      SY.lastHit = now; SY.env = Math.max(SY.env, kick);
      const a = Math.random() * TAU; SY.sx = Math.cos(a); SY.sy = Math.sin(a);
      if (kick > 0.78 || SY.drop > 0.5) SY.flash = Math.max(SY.flash, 0.55 + 0.45 * kick);
    }
    SY.prevKick = kick; SY.env *= Math.exp(-dt / (spb * 0.32)); SY.flash *= Math.exp(-dt / 0.14);
    // энергия: медленная (секция песни) и быстрая (сейчас); резкий рост = дроп / припев
    const en = o.energy != null ? o.energy : Math.min(1, level * 2.2);
    SY.fast += (en - SY.fast) * Math.min(1, dt * 6); SY.slow += (en - SY.slow) * Math.min(1, dt * 0.25);
    const surge = o.playing ? Math.max(0, SY.fast / Math.max(0.08, SY.slow) - 1.25) : 0;
    if (surge > 0.35 && SY.drop < 0.2) { SY.drop = 1; HUE = (HUE + 0.09) % 1; SY.flash = Math.max(SY.flash, 0.9); }
    SY.drop *= Math.exp(-dt / 1.6);
    const energy = Math.max(0, Math.min(1, SY.fast));
    const trails = o.mode === 'visual' ? (FX.trails || 0) / 100 : 0;
    // 2.0.11: в «Обложке» без полос и в «Видео» холст пустой — раньше его всё равно стирали каждый кадр, и браузер
    // каждый кадр заново отправлял на видеокарту полноэкранный прозрачный слой (на слабой графике −60…70% FPS)
    if (o.mode === 'video' || (o.mode === 'cover' && o.noBars)) { if (!blankCv) { ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.clearRect(0, 0, cv.width, cv.height); ctx.restore(); blankCv = true; cv.style.visibility = 'hidden'; } return; }
    if (blankCv) { blankCv = false; cv.style.visibility = ''; } // 2.0.11: пустой холст ещё и прячем — браузер не смешивает его с экраном
    if (trails > 0) { ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.globalCompositeOperation = 'destination-out'; ctx.fillStyle = `rgba(0,0,0,${(1 - trails) * 0.9 + 0.05})`; ctx.fillRect(0, 0, cv.width, cv.height); ctx.restore(); }
    else ctx.clearRect(0, 0, W, H);
    if (o.mode === 'video') return;
    const a = o.artRect, C = o.center, cx = a ? a.x : C ? C.x : W / 2, cy = a ? a.y : C ? C.y : H * 0.45, R = a ? a.r : Math.min(W, H) * 0.16;
    const g = {
      ctx, W, H, cx, cy, R, dt, t: now / 1000, bands, peaks, NB, wave: o.wave, playing: !!o.playing, beat, kick: o.beat || 0, pulse,
      bpm, pos, phase, bar: ((pos / 4) % 1 + 1) % 1, beatN: Math.floor(pos), tick: Math.floor(pos) + ease(phase), speed: bpm / 120, conf: T.conf || 0,
      level, energy, drop: SY.drop, hit: SY.env, bass: bass(), mid: mid(), high: high(), band, mix, col, rgba, rr, smoothClosed, glow, noGlow, ease, TAU, opts: o.opts || {}, intensity: o.intensity || 1, cover: !!a
    };
    if (o.mode === 'cover') { if (!o.noBars) coverBars(g); return; }
    const S = REG.find((x) => x.id === o.style) || (EXT && EXT().find((x) => x.id === o.style)) || REG[0]; if (!S) return;
    g.state = state[S.id] || (state[S.id] = {}); g.lw = LW; g.dens = DENS; g.coverImg = COVER;
    const c0 = performance.now();
    const zoom = (FX.zoom ?? 30) / 100, shake = (FX.shake || 0) / 100;
    try {
      ctx.save();
      if (zoom > 0 || shake > 0) {
        // зум «вдыхает» на реальном ударе и сильнее в громких частях; тряска — один толчок в сторону и плавный возврат
        const hit = Math.max(SY.env, pulse * 0.35), dyn = 0.55 + energy * 0.6 + SY.drop * 0.5;
        const k = 1 + hit * hit * zoom * 0.085 * dyn, amp = shake * SY.env * SY.env * (14 + g.bass * 22) * dyn;
        const sx = amp * SY.sx, sy = amp * SY.sy;
        ctx.translate(cx + sx, cy + sy); ctx.scale(k, k); ctx.translate(-cx, -cy);
      }
      // 2.0.6: 3D-стили — на видеокарте; если WebGL недоступен или стиль упал на GPU — обычный canvas
      const R = S.gl && !S.glFail ? gpuFor() : null;
      if (R && R.begin(W, H, cv.width / Math.max(1, W))) { try { S.gl(g, R); ctx.drawImage(gcv, 0, 0, W, H); } catch (e) { console.warn('viz gpu', S.id, e); S.glFail = true; S.draw(g); } }
      else S.draw(g);
    } catch (e) { console.error('viz', S.id, e); } finally { ctx.restore(); ctx.shadowBlur = 0; ctx.globalAlpha = 1; ctx.globalCompositeOperation = 'source-over'; }
    const mir = FX.mirror || 'off';
    if (mir !== 'off') {
      ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0); const w2 = cv.width / 2, h2 = cv.height / 2;
      if (mir === 'h' || mir === 'quad') { ctx.clearRect(w2, 0, cv.width - w2, cv.height); ctx.translate(cv.width, 0); ctx.scale(-1, 1); ctx.drawImage(cv, 0, 0, w2, cv.height, 0, 0, w2, cv.height); ctx.setTransform(1, 0, 0, 1, 0, 0); }
      if (mir === 'v' || mir === 'quad') { ctx.clearRect(0, h2, cv.width, cv.height - h2); ctx.translate(0, cv.height); ctx.scale(1, -1); ctx.drawImage(cv, 0, 0, cv.width, h2, 0, 0, cv.width, h2); }
      ctx.restore();
    }
    const fl = (FX.flash || 0) / 100;
    // вспышка — на сильных ударах и на дропе, а не просто на каждой четвёртой доле
    if (fl > 0 && SY.flash > 0.02 && o.playing) { ctx.save(); ctx.globalCompositeOperation = 'lighter'; const gr = ctx.createRadialGradient(cx, cy, 0, cx, cy, Math.max(W, H) * 0.75); gr.addColorStop(0, mix(g.bar, Math.min(0.5, fl * 0.3 * SY.flash))); gr.addColorStop(1, mix(g.bar + 0.3, fl * 0.08 * SY.flash)); ctx.fillStyle = gr; ctx.fillRect(0, 0, W, H); ctx.restore(); }
    // авто-качество: если кадр рисуется слишком долго — выключаем свечение и уменьшаем разрешение холста
    cost = cost * 0.95 + (performance.now() - c0) * 0.05;
    // 2.0.4: сторож частоты кадров из 2.0.3 убран — он ошибочно выключал свечение и снижал чёткость, хотя тормозил не визуал
    if (!autoLow && cost > 9) { autoLow = true; goodSince = now; flips++; }
    else if (autoLow && flips < 3) { if (cost > 2) goodSince = now; else if (now - goodSince > 20000) { autoLow = false; cost = 0; } }
  }
  function coverBars(g) {
    const n = 48, gap = 3, bw = Math.max(2, (W * 0.8) / (n * 2) - gap), x0 = W / 2, base = H - 4, mh = H * 0.13;
    for (let i = 0; i < n; i++) {
      const h = 3 + Math.pow(bands[i], 1.25) * mh;
      ctx.fillStyle = mix(i / n, 0.22);
      for (const s of [-1, 1]) { const x = x0 + s * (i * (bw + gap) + gap / 2) - (s < 0 ? bw : 0); rr(x, base - h, bw, h, bw / 2); ctx.fill(); }
    }
  }

  // ---------- встроенные стили ----------
  const B = [];
  B.push({ id: 'ring', name: 'Кольцо', icon: 'v-ring', cover: 'show', draw(g) {
    const { ctx, cx, cy } = g, R = g.R * 1.02, N = 128, rot = g.tick * (TAU / 64);
    const gr = ctx.createRadialGradient(cx, cy, R * 0.8, cx, cy, R * (1.9 + g.bass * 0.5));
    gr.addColorStop(0, g.mix(0.1, 0)); gr.addColorStop(0.4, g.mix(0.2, 0.12 + g.beat * 0.2)); gr.addColorStop(1, g.mix(0.3, 0));
    ctx.fillStyle = gr; ctx.beginPath(); ctx.arc(cx, cy, R * 2.4, 0, TAU); ctx.fill();
    ctx.lineCap = 'round'; ctx.lineWidth = (Math.max(2, ((TAU * R) / N) * 0.48)) * LW;
    for (let i = 0; i < N; i++) {
      const m = i < N / 2 ? i : N - 1 - i, v = bands[Math.floor((m / (N / 2)) * 56)];
      const len = 3 + Math.pow(v, 1.4) * R * 0.7 * (1 + g.beat * 0.3);
      const a = rot + (i / N) * TAU - Math.PI / 2, c = Math.cos(a), s = Math.sin(a), r0 = R + 8;
      ctx.strokeStyle = g.mix(i / N + g.pos * 0.02, 0.95);
      ctx.beginPath(); ctx.moveTo(cx + c * r0, cy + s * r0); ctx.lineTo(cx + c * (r0 + len), cy + s * (r0 + len)); ctx.stroke();
    }
    const st = g.state; st.sh = st.sh || []; if (g.beatN !== st.lb && g.playing) { st.lb = g.beatN; st.sh.push({ r: R + 8, a: 0.3 + g.beat * 0.4 }); }
    for (let i = st.sh.length - 1; i >= 0; i--) { const s = st.sh[i]; s.r += g.dt * R * 1.6 * g.speed; s.a -= g.dt * 0.9; if (s.a <= 0) { st.sh.splice(i, 1); continue; } ctx.strokeStyle = g.mix(s.r / 400, s.a); ctx.lineWidth = (2) * LW; ctx.beginPath(); ctx.arc(cx, cy, s.r, 0, TAU); ctx.stroke(); }
  } });
  B.push({ id: 'pulse', name: 'Пульс', icon: 'v-pulse', cover: 'show', draw(g) {
    const { ctx, cx, cy, R } = g, st = g.state; st.rings = st.rings || [];
    if (g.beatN !== st.lb && g.playing) { st.lb = g.beatN; const down = g.beatN % 4 === 0; st.rings.push({ r: R, a: down ? 0.9 : 0.5, w: down ? 5 : 2.5, c: g.pos * 0.05 }); }
    ctx.globalCompositeOperation = 'lighter';
    for (let i = st.rings.length - 1; i >= 0; i--) { const s = st.rings[i]; s.r += g.dt * (R * 1.3) * g.speed * (1 + s.w / 5); s.a -= g.dt * 0.55 * g.speed; if (s.a <= 0) { st.rings.splice(i, 1); continue; } ctx.strokeStyle = g.mix(s.c + s.r / 900, s.a); ctx.lineWidth = (s.w) * LW; ctx.beginPath(); ctx.arc(cx, cy, s.r, 0, TAU); ctx.stroke(); }
    const r2 = R * (1.08 + g.beat * 0.18 + g.bass * 0.15);
    const gr = ctx.createRadialGradient(cx, cy, R * 0.9, cx, cy, r2 * 1.6); gr.addColorStop(0, g.mix(0, 0.5 * (0.4 + g.beat))); gr.addColorStop(1, g.mix(0.4, 0));
    ctx.fillStyle = gr; ctx.beginPath(); ctx.arc(cx, cy, r2 * 1.6, 0, TAU); ctx.fill();
  } });
  B.push({ id: 'blob', name: 'Жидкость', icon: 'v-blob', cover: 'show', draw(g) {
    const { ctx, cx, cy, R } = g, N = 48;
    for (let L = 2; L >= 0; L--) {
      const pts = [];
      for (let i = 0; i < N; i++) {
        const a = (i / N) * TAU, m = i < N / 2 ? i / (N / 2) : (N - i) / (N / 2);
        const n = Math.sin(a * 3 + g.pos * 0.5 + L) * 0.5 + Math.sin(a * 5 - g.pos * 0.33 + L * 2) * 0.5;
        const r = R * (1.12 + L * 0.16) + g.band(m * 0.7) * R * (0.5 + L * 0.2) + n * R * 0.05 * (1 + g.level * 3) + g.beat * R * 0.08;
        pts.push([cx + Math.cos(a) * r, cy + Math.sin(a) * r]);
      }
      g.smoothClosed(pts);
      const gr = ctx.createRadialGradient(cx, cy, R * 0.8, cx, cy, R * (2 + L * 0.4)); gr.addColorStop(0, g.mix(L / 3 + g.pos * 0.01, 0.55 - L * 0.12)); gr.addColorStop(1, g.mix(L / 3 + 0.3, 0.05));
      ctx.fillStyle = gr; ctx.fill();
    }
  } });
  B.push({ id: 'sun', name: 'Солнце', icon: 'v-sun', cover: 'show', draw(g) {
    const { ctx, cx, cy, R } = g, N = 96, rot = g.tick * (TAU / 96);
    ctx.globalCompositeOperation = 'lighter';
    for (let L = 0; L < 2; L++) {
      const pts = [];
      for (let i = 0; i < N; i++) { const m = i < N / 2 ? i / (N / 2) : (N - i) / (N / 2), a = rot * (L ? -1 : 1) + (i / N) * TAU; const v = Math.pow(g.band(m * 0.85), 1.2); const r = R * 1.06 + v * R * (0.9 - L * 0.3) * (1 + g.beat * 0.25); pts.push([cx + Math.cos(a) * r, cy + Math.sin(a) * r]); }
      g.smoothClosed(pts); ctx.fillStyle = g.mix(L * 0.4 + g.pos * 0.015, 0.28); ctx.fill(); ctx.strokeStyle = g.mix(L * 0.4 + 0.1, 0.8); ctx.lineWidth = (1.6) * LW; ctx.stroke();
    }
  } });
  B.push({ id: 'bars', name: 'Эквалайзер', icon: 'v-bars', cover: 'hide', draw(g) {
    const { ctx, W, H } = g, n = 64, gap = 3, bw = Math.max(2, (W * 0.86) / (n * 2) - gap), x0 = W / 2, base = H * 0.62, mh = H * 0.42;
    for (let i = 0; i < n; i++) {
      const v = Math.pow(bands[i], 1.25), h = 3 + v * mh * (1 + g.beat * 0.12), pk = 3 + Math.pow(peaks[i], 1.25) * mh;
      for (const s of [-1, 1]) {
        const x = x0 + s * (i * (bw + gap) + gap / 2) - (s < 0 ? bw : 0);

        ctx.fillStyle = `rgba(${Math.min(255, d[o + 4] * lit * 255) | 0},${Math.min(255, d[o + 5] * lit * 255) | 0},${Math.min(255, d[o + 6] * lit * 255) | 0},${Math.min(1, (0.55 + d[o + 2] * 0.35 + a * 0.4) * intro).toFixed(3)})`;
        ctx.fillRect(c.ox + xr * kk - s / 2, c.oy - yr * kk - s / 2, s, s);
      }
    } });

  // ---------- 2.0.6: те же 3D-стили на видеокарте (gpu.js). Проекция, толщина линий и свечение — в шейдерах ----------
  const c255 = (f) => { const c = col(f); return [c[0] / 255, c[1] / 255, c[2] / 255]; };
  const camOf = (yaw, pitch, dist, f, ox, oy, roll) => ({ yaw, pitch, dist, f, ox, oy, roll: roll || 0, lw: LW, glow: lowQ ? 0 : GLOW });
  // глубина точки для сортировки «дальние — первыми» (как в 2D-версии)
  const depthOf = (yaw, pitch, dist) => { const cy0 = Math.cos(yaw), sy0 = Math.sin(yaw), cp = Math.cos(pitch), sp = Math.sin(pitch); return (x, y, z) => y * sp + (x * sy0 + z * cy0) * cp + dist; };
  const GL3D = {
    city(g, R) {
      const { W, H } = g, st = g.state, N = 12;
      st.yaw = (st.yaw || 0) + g.dt * (0.12 + g.bass * 0.25) * g.speed;
      const pitch = -(0.55 + Math.sin(g.t * 0.11) * 0.14), dist = 17 + Math.sin(g.t * 0.07) * 2.5;
      R.camera(camOf(st.yaw, pitch, dist, Math.min(W, H) * 0.95, W / 2, H * 0.6));
      let b = R.buf((N + 1) * 2 * 14), n = 0; const gc = c255(0.6);
      const seg = (ax, az, bx, bz) => { b.set([ax, 0, az, bx, 0, bz, 0, 1, 0, 0, gc[0], gc[1], gc[2], 0.12], n * 14); n++; };
      for (let i = 0; i <= N; i++) { seg(i - N / 2, -N / 2, i - N / 2, N / 2); seg(-N / 2, i - N / 2, N / 2, i - N / 2); }
      R.draw('sg', b, n, 'max');
      const Z = depthOf(st.yaw, pitch, dist), bars = [];
      for (let i = 0; i < N; i++) for (let j = 0; j < N; j++) { const x = i - N / 2 + 0.5, z = j - N / 2 + 0.5, d = Math.hypot(x, z) / (N * 0.71); const v = g.band(Math.min(0.95, d * 0.85)); const h = 0.12 + Math.pow(v, 1.3) * 5 * (1 + g.beat * 0.35 * (1 - d)); bars.push([x, z, h, d, v, Z(x, 0, z)]); }
      bars.sort((a, c) => c[5] - a[5]);
      // ореолы ярких зданий (вместо shadowBlur), потом сами здания
      b = R.buf(bars.length * 14); n = 0;
      for (const [x, z, h, d, v] of bars) { if (v <= 0.55) continue; const c = c255(d * 0.9 + g.pos * 0.01); b.set([x, 0, z, x, h, z, 0.62, 1.5, 14 * v, 1, c[0], c[1], c[2], 0.35 + v * 0.65], n * 14); n++; }
      R.draw('sg', b, n);
      b = R.buf(bars.length * 9); n = 0;
      for (const [x, z, h, d, v] of bars) { const c = c255(d * 0.9 + g.pos * 0.01); b.set([x, z, h, 0.5, c[0], c[1], c[2], 0.35 + v * 0.65, 0.25 + v * 0.5], n * 9); n++; }
      R.draw('bx', b, n);
    },
    flight(g, R) {
      const { W, H } = g, st = g.state, C = 40, ROWS = 34;
      st.rows = st.rows || Array.from({ length: ROWS }, () => new Float32Array(C)); st.acc = (st.acc || 0) + g.dt * (5 + g.bass * 9) * g.speed;
      while (st.acc >= 1) { st.acc -= 1; const r = new Float32Array(C); for (let i = 0; i < C; i++) { const m = Math.abs(i - (C - 1) / 2) / ((C - 1) / 2); r[i] = Math.pow(g.band(Math.min(0.95, 0.04 + m * 0.6)), 1.2) * (0.6 + m * 1.4); } st.rows.unshift(r); if (st.rows.length > ROWS) st.rows.pop(); }
      const roll = Math.sin(g.t * 0.23) * 0.07 + Math.sin(g.t * 0.07) * 0.05;
      R.camera(camOf(Math.sin(g.t * 0.09) * 0.25, -(0.3 + Math.sin(g.t * 0.13) * 0.06), 2.2, Math.min(W, H) * 0.9, W / 2, H * 0.78, roll));
      const b = R.buf(st.rows.length * (C - 1) * 14); let n = 0;
      for (let k = st.rows.length - 1; k >= 0; k--) {
        const r = st.rows[k], z = (k + st.acc - 1) * 0.9, fade = 1 - k / ROWS, c = c255(k / ROWS + g.bar * 0.2), a = 0.15 + fade * 0.85, w = 0.6 + fade * 2.2, gl = k < 4 ? 16 : 0;
        for (let i = 0; i < C - 1; i++) { b.set([(i - (C - 1) / 2) * 0.55, r[i] * 3.2, z, (i + 1 - (C - 1) / 2) * 0.55, r[i + 1] * 3.2, z, 0, w, gl, 0, c[0], c[1], c[2], a], n * 14); n++; }
      }
      R.draw('sg', b, n, 'max');
    },
    orbits(g, R) {
      const { W, H } = g, st = g.state; st.yaw = (st.yaw || 0) + g.dt * 0.22 * g.speed;
      const pitch = 0.35 + Math.sin(g.t * 0.17) * 0.3; R.camera(camOf(st.yaw, pitch, 9, Math.min(W, H) * 0.95, g.cx, g.cy));
      const Z = depthOf(st.yaw, pitch, 9), rings = 4, n0 = 90, pts = [];
      for (let k = 0; k < rings; k++) {
        const tilt = 0.4 + k * 0.55 + Math.sin(g.t * 0.1 + k) * 0.2, spin = g.tick * 0.05 * (k % 2 ? -1 : 1) + k, ct = Math.cos(tilt), sn = Math.sin(tilt);
        for (let i = 0; i < n0; i++) { const a = spin + (i / n0) * TAU, v = g.band(((i < n0 / 2 ? i : n0 - i) / (n0 / 2)) * 0.7 + k * 0.05), r = 1.6 + k * 0.85 + v * 1.1 * (1 + g.beat * 0.3); const x = Math.cos(a) * r, y = Math.sin(a) * r * ct, z = Math.sin(a) * r * sn; pts.push([x, y, z, k, v, Z(x, y, z)]); }
      }
      pts.sort((a, c) => c[5] - a[5]);
      const b = R.buf(pts.length * 11); let n = 0;
      for (const [x, y, z, k, v] of pts) { const c = c255(k / rings + v * 0.3); b.set([x, y, z, 0.05 + v * 0.09, 0.8, c[0], c[1], c[2], 0.4 + v * 0.6, v > 0.5 ? 12 : 0, 0], n * 11); n++; }
      R.draw('pt', b, n);
    },
    helix(g, R) {
      const { W, H } = g, st = g.state; st.yaw = (st.yaw || 0) + g.dt * (0.35 + g.bass * 0.6) * g.speed;
      R.camera(camOf(st.yaw, Math.sin(g.t * 0.12) * 0.35, 8, Math.min(W, H) * 0.9, W / 2, H / 2));
      const n0 = 70, P = [];
      for (let i = 0; i < n0; i++) { const u = i / (n0 - 1), y = (u - 0.5) * 9, v = g.band(Math.min(0.95, u * 0.9)), r = 1.2 + v * 1.6 * (1 + g.beat * 0.3), a = u * TAU * 2.2 + g.tick * 0.08; P.push([Math.cos(a) * r, y, Math.sin(a) * r, u, v]); }
      let b = R.buf(n0 * 14), n = 0;
      for (let i = 0; i < n0; i += 3) { const [x, y, z, u, v] = P[i], c = c255(u); b.set([x, y, z, -x, y, -z, 0, 1.2, 0, 0, c[0], c[1], c[2], 0.12 + v * 0.35], n * 14); n++; }
      R.draw('sg', b, n, 'max');
      b = R.buf(n0 * 2 * 11); n = 0;
      for (const s of [0, 1]) for (const [x, y, z, u, v] of P) { const c = c255(u + s * 0.5), k = s ? -1 : 1; b.set([x * k, y, z * k, 0.06 + v * 0.1, 1, c[0], c[1], c[2], 0.5 + v * 0.5, v > 0.55 ? 14 : 0, 0], n * 11); n++; }
      R.draw('pt', b, n);
    },
  };
  B.forEach((s) => { if (GL3D[s.id]) s.gl = GL3D[s.id]; });
  B.forEach((s) => REG.push(s));

  function register(s) { if (!s || !s.id || typeof s.draw !== 'function') throw new Error('Viz.register: нужен id и draw()'); const i = REG.findIndex((x) => x.id === s.id); s.cover = s.cover || 'hide'; if (i >= 0) REG[i] = s; else REG.push(s); if (onChange) onChange(); }

  // палитра из обложки: гистограмма оттенков, яркие насыщенные цвета
  function palette(src) {
    return new Promise((res) => {
      const img = new Image(); img.onload = () => {
        try {
          const c = document.createElement('canvas'); c.width = c.height = 48; const x = c.getContext('2d');
          x.drawImage(img, 0, 0, 48, 48); const d = x.getImageData(0, 0, 48, 48).data;
          const bins = Array.from({ length: 18 }, () => ({ w: 0, r: 0, g: 0, b: 0 }));
          for (let i = 0; i < d.length; i += 4) {
            const r = d[i], g = d[i + 1], b = d[i + 2], mx = Math.max(r, g, b), mn = Math.min(r, g, b), s = mx ? (mx - mn) / mx : 0, v = mx / 255;
            if (s < 0.25 || v < 0.2) continue;
            let h = 0; if (mx !== mn) { h = mx === r ? (g - b) / (mx - mn) : mx === g ? 2 + (b - r) / (mx - mn) : 4 + (r - g) / (mx - mn); h = ((h * 60) + 360) % 360; }
            const bi = bins[Math.floor(h / 20)], w = s * s * v; bi.w += w; bi.r += r * w; bi.g += g * w; bi.b += b * w;
          }
          const top = bins.map((b, i) => ({ ...b, i })).filter((b) => b.w > 0).sort((a, b) => b.w - a.w);
          const pick = []; for (const b of top) { if (pick.every((p) => Math.min(Math.abs(p.i - b.i), 18 - Math.abs(p.i - b.i)) >= 3)) pick.push(b); if (pick.length === 3) break; }
          if (!pick.length) return res(null);
          const boost = (b) => { const r = b.r / b.w, g = b.g / b.w, bl = b.b / b.w; const k = 235 / (Math.max(r, g, bl) || 1); return [Math.min(255, r * k), Math.min(255, g * k), Math.min(255, bl * k)]; };
          const out = pick.map(boost); while (out.length < 3) { const p = out[out.length - 1]; out.push([p[2], p[0], p[1]]); }
          res(out);
        } catch { res(null); }
      };
      img.onerror = () => res(null); img.src = src;
    });
  }
  return { init, destroy, draw, palette, resize, register, setDpr: (v) => { if (opts.maxDpr !== v) { opts.maxDpr = v; resize(); } }, isLow: () => lowQ, setGpu: (v) => { opts.gpu = v !== false; }, gpuOn: () => !!(G && G.ok()), setReact: (r) => (REACT = Math.max(0, Math.min(1, r))), setPalette: (p) => (pal = p), get STYLES() { return REG; }, onChange: (f) => (onChange = f), bands: () => bands };
};
